CModule 3: C Data Types & Variables Implementation Sample
#include <stdio.h>
#include <stdint.h>
#include <stdbool.h>

/* Structure padding & memory layout inspection */
struct SensorPacket {
    uint8_t  sensor_id;   /* 1 byte */
    /* 3 bytes padding */
    uint32_t timestamp;   /* 4 bytes */
    uint16_t reading_raw; /* 2 bytes */
    /* 2 bytes padding */
};

int main(void) {
    printf("Size of SensorPacket: %zu bytes (expected 12 with alignment)
",
           sizeof(struct SensorPacket));
    return 0;
}
