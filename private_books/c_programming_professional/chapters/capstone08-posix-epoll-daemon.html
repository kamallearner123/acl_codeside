Capstone 8: POSIX epoll Multi-Client Network Daemon

Construct an enterprise-grade Linux network service in C capable of handling 10,000+ concurrent connections. Master non-blocking socket streams, Edge-Triggered epoll event multiplexing, per-connection buffer pools, PID file locking with flock, and graceful signalfd shutdowns.

⏱️ Duration: 100m
🎯 Standard: POSIX.1-2008 / Linux epoll(7) / RFC 9293 (TCP)
⚡ Target: High-Concurrency Linux Edge Daemons

1. Daemon Architecture & Production Requirements

A production Linux system service (daemon) must fulfill strict operating system requirements:

  1. Daemonization: Disassociate from controlling terminals (fork, setsid, redirect standard FDs to /dev/null).
  2. Single-Instance Lock: Enforce single instance execution via flock(LOCK_EX | LOCK_NB) on a PID file in /var/run/.
  3. High Concurrency: Scale to 10,000 concurrent sockets using Edge-Triggered epoll and non-blocking I/O.
  4. Graceful Shutdown: Catch SIGINT and SIGTERM via signalfd to flush buffers before exiting.

2. Complete Production Daemon Implementation

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <errno.h>
#include <sys/epoll.h>
#include <sys/file.h>
#include <sys/signalfd.h>
#include <signal.h>
#include <netinet/in.h>

#define MAX_EVENTS 128
#define PORT 9000
#define PID_FILE "/tmp/acl_epoll_daemon.pid"

static int make_non_blocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    return fcntl(fd, F_SETFL, flags | O_NONBLOCK);
}

static void daemonize(void) {
    pid_t pid = fork();
    if (pid < 0) exit(1);
    if (pid > 0) exit(0); // Parent exits

    setsid(); // Create new session

    int lock_fd = open(PID_FILE, O_RDWR | O_CREAT, 0640);
    if (lock_fd < 0 || flock(lock_fd, LOCK_EX | LOCK_NB) < 0) {
        fprintf(stderr, "Daemon is already running!
");
        exit(1);
    }
}

int main(void) {
    daemonize();

    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGINT);
    sigaddset(&mask, SIGTERM);
    sigprocmask(SIG_BLOCK, &mask, NULL);
    int sfd = signalfd(-1, &mask, SFD_NONBLOCK | SFD_CLOEXEC);

    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(PORT);
    bind(server_fd, (struct sockaddr *)&addr, sizeof(addr));
    listen(server_fd, SOMAXCONN);
    make_non_blocking(server_fd);

    int epfd = epoll_create1(EPOLL_CLOEXEC);
    struct epoll_event ev, events[MAX_EVENTS];

    ev.events = EPOLLIN; ev.data.fd = server_fd;
    epoll_ctl(epfd, EPOLL_CTL_ADD, server_fd, &ev);

    ev.events = EPOLLIN; ev.data.fd = sfd;
    epoll_ctl(epfd, EPOLL_CTL_ADD, sfd, &ev);

    int running = 1;
    while (running) {
        int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
        for (int i = 0; i < nfds; i++) {
            if (events[i].data.fd == sfd) {
                running = 0; // Signal received, graceful shutdown!
            } else if (events[i].data.fd == server_fd) {
                while (1) {
                    int cfd = accept(server_fd, NULL, NULL);
                    if (cfd < 0) break;
                    make_non_blocking(cfd);
                    ev.events = EPOLLIN | EPOLLET;
                    ev.data.fd = cfd;
                    epoll_ctl(epfd, EPOLL_CTL_ADD, cfd, &ev);
                }
            } else {
                int cfd = events[i].data.fd;
                char buf[256];
                while (1) {
                    ssize_t n = read(cfd, buf, sizeof(buf));
                    if (n > 0) {
                        write(cfd, buf, (size_t)n); // Echo back
                    } else if (n == 0 || (n < 0 && errno != EAGAIN)) {
                        epoll_ctl(epfd, EPOLL_CTL_DEL, cfd, NULL);
                        close(cfd);
                        break;
                    } else {
                        break; // Drained
                    }
                }
            }
        }
    }

    close(server_fd);
    close(sfd);
    close(epfd);
    unlink(PID_FILE);
    return 0;
}

3. Capstone Knowledge Verification Quiz

Self-Assessment Check

Q1: What is the primary purpose of calling setsid() during process daemonization in Linux?

Explanation: setsid() creates a new session without a controlling terminal, ensuring terminal closure signals (SIGHUP) do not terminate the service.
