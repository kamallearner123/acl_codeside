Capstone 5: Embedded CLI Shell with Tab Completion

Design an industrial, non-blocking serial Command Line Interface (CLI) for embedded firmware. Master VT100 ANSI terminal escape sequence parsing, command table dispatchers, tab auto-completion, argument tokenization, and command history buffers.

⏱️ Duration: 80m
🎯 Standard: VT100 / ANSI X3.64 / MISRA C Aligned
⚡ Target: Embedded Serial Diagnostic Consoles

1. The Role of Serial Diagnostic Shells in Production

A production device deployed in the field cannot rely on JTAG debuggers for maintenance. An interactive, non-blocking Command Line Interface (CLI) over serial or USB CDC allows field engineers to inspect runtime telemetry, read error logs, trigger self-tests, and calibrate sensors without halting the real-time control loops.

2. Production Embedded CLI Implementation in C

#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define CLI_MAX_INPUT_LEN   64U
#define CLI_MAX_ARGS        8U

typedef int (*cli_cmd_fn)(int argc, char **argv);

typedef struct {
    const char *name;
    const char *help;
    cli_cmd_fn  handler;
} cli_command_t;

/* Forward declarations */
static int cmd_help(int argc, char **argv);
static int cmd_version(int argc, char **argv);
static int cmd_read_adc(int argc, char **argv);

static const cli_command_t g_commands[] = {
    {"help",    "Display available commands", cmd_help},
    {"version", "Print firmware build information", cmd_version},
    {"adc",     "Read analog channel: adc <ch>", cmd_read_adc},
};
#define NUM_COMMANDS (sizeof(g_commands) / sizeof(g_commands[0]))

static char     g_input_buf[CLI_MAX_INPUT_LEN];
static uint32_t g_buf_idx = 0;

static void cli_put_char(char c) {
    /* Send character over UART (Echo back to terminal) */
    putchar(c);
    fflush(stdout);
}

static void cli_put_string(const char *str) {
    while (*str) cli_put_char(*str++);
}

static int cmd_help(int argc, char **argv) {
    (void)argc; (void)argv;
    cli_put_string("
Available Commands:
");
    for (size_t i = 0; i < NUM_COMMANDS; i++) {
        char line[128];
        snprintf(line, sizeof(line), "  %-10s - %s
", g_commands[i].name, g_commands[i].help);
        cli_put_string(line);
    }
    return 0;
}

static int cmd_version(int argc, char **argv) {
    (void)argc; (void)argv;
    cli_put_string("
Firmware Version: 2.4.1 (Git: a8f9c1b) Built: " __DATE__ "
");
    return 0;
}

static int cmd_read_adc(int argc, char **argv) {
    if (argc < 2) {
        cli_put_string("
Usage: adc <channel_0_to_7>
");
        return -1;
    }
    cli_put_string("
[ADC] Channel reading: 3.298 V
");
    return 0;
}

/* Dispatch command */
static void cli_execute_line(char *line) {
    char *argv[CLI_MAX_ARGS];
    int argc = 0;

    char *token = strtok(line, " 	");
    while (token && argc < CLI_MAX_ARGS) {
        argv[argc++] = token;
        token = strtok(NULL, " 	");
    }

    if (argc == 0) return;

    for (size_t i = 0; i < NUM_COMMANDS; i++) {
        if (strcmp(argv[0], g_commands[i].name) == 0) {
            g_commands[i].handler(argc, argv);
            return;
        }
    }
    cli_put_string("
Unknown command. Type 'help'.
");
}

/* Tab auto-completion */
static void cli_handle_tab(void) {
    if (g_buf_idx == 0) return;
    for (size_t i = 0; i < NUM_COMMANDS; i++) {
        if (strncmp(g_commands[i].name, g_input_buf, g_buf_idx) == 0) {
            /* Found match! Auto-complete remaining characters */
            const char *suffix = g_commands[i].name + g_buf_idx;
            while (*suffix) {
                g_input_buf[g_buf_idx++] = *suffix;
                cli_put_char(*suffix++);
            }
            break;
        }
    }
}

/* Character Ingestion State Machine */
void cli_feed_char(char c) {
    if (c == '
' || c == '
') {
        cli_put_string("
");
        g_input_buf[g_buf_idx] = '';
        cli_execute_line(g_input_buf);
        g_buf_idx = 0;
        cli_put_string("prompt> ");
    } else if (c == '	') {
        cli_handle_tab();
    } else if (c == 0x08 || c == 0x7F) { // Backspace
        if (g_buf_idx > 0) {
            g_buf_idx--;
            cli_put_string(" "); // Erase character visually
        }
    } else if (c >= 32 && c <= 126) {
        if (g_buf_idx < (CLI_MAX_INPUT_LEN - 1)) {
            g_input_buf[g_buf_idx++] = c;
            cli_put_char(c);
        }
    }
}

Self-Assessment Check

Q1: What ASCII character sequence does an ANSI/VT100 terminal emit when the Backspace key is pressed?

Explanation: Terminal emulators transmit either BS (0x08) or DEL (0x7F) for backspace; sending   back to the terminal visually erases the character.
