Module 9: Function Pointers, Callbacks & Event Dispatchers

Harness function pointers for inversion of control, O(1) jump tables, simulated C++ vtables for swap-in device drivers, and real-time CAN bus message router engines.

⏱️ Duration: 50m
🎯 System: Memory Layout, Bus Faults, MISRA C:2012
⚡ Architecture: ARM Cortex-M / Linux x86_64

9.1 Function Pointer Syntax, Declarations & Typedefs

Functions in C reside in the .text (Flash/Code) memory segment. A function pointer stores the entry address of that executable block. Because raw C function pointer syntax is notoriously difficult to read, always use typedef:

Cfunction_pointer_syntax.c
#include <stdint.h>
#include <stdio.h>

/* Raw syntax: void (*callback)(uint8_t event_id, void *context); */

/* Clean, professional engineering syntax: */
typedef void (*EventCallback_t)(uint8_t event_id, void *user_context);

void on_button_press(uint8_t event_id, void *user_context) {
    printf("Button pressed: Event 0x%02X, Context: %p\n", event_id, user_context);
}

int main(void) {
    EventCallback_t handler = on_button_press;
    handler(0x01, NULL); /* Invocation */
    return 0;
}

9.2 Jump Tables for O(1) State Machine & Command Dispatch

Large switch-case statements with dozens of branches produce long chains of conditional comparison instructions (cmp / bne). In deterministic systems, a jump table array of function pointers delivers immediate O(1) constant-time execution:

Cjump_table_dispatcher.c
#include <stdint.h>
#include <stdbool.h>

typedef enum {
    CMD_SYSTEM_RESET   = 0,
    CMD_START_TELEMETRY = 1,
    CMD_STOP_TELEMETRY  = 2,
    CMD_CALIBRATE_IMU   = 3,
    CMD_MAX_COUNT
} CommandId_t;

typedef void (*CommandHandler_t)(const uint8_t *payload, uint16_t length);

static void handle_reset(const uint8_t *payload, uint16_t length)       { /* Reset HW */ }
static void handle_start_telem(const uint8_t *payload, uint16_t length) { /* Enable DMA */ }
static void handle_stop_telem(const uint8_t *payload, uint16_t length)  { /* Disable DMA */ }
static void handle_calibrate(const uint8_t *payload, uint16_t length)   { /* Calibrate */ }

/* Direct O(1) Jump Table stored in Read-Only Flash (.rodata) */
static const CommandHandler_t CMD_TABLE[CMD_MAX_COUNT] = {
    [CMD_SYSTEM_RESET]    = handle_reset,
    [CMD_START_TELEMETRY] = handle_start_telem,
    [CMD_STOP_TELEMETRY]  = handle_stop_telem,
    [CMD_CALIBRATE_IMU]   = handle_calibrate
};

void dispatch_command(uint8_t cmd_id, const uint8_t *payload, uint16_t len) {
    if (cmd_id < (uint8_t)CMD_MAX_COUNT) {
        CommandHandler_t fn = CMD_TABLE[cmd_id];
        if (fn != NULL) {
            fn(payload, len); /* Direct indirect call: BLX R3 */
        }
    }
}

9.3 Simulating Polymorphism & Vtables in Pure C

In embedded software architecture, you frequently need to support interchangeable device drivers (e.g., swapping an I2C Flash memory chip for an SPI EEPROM without altering application code). This is solved through a Driver Interface Vtable Struct:

Cnvram_driver_interface.h
#ifndef NVRAM_DRIVER_H
#define NVRAM_DRIVER_H

#include <stdint.h>
#include <stddef.h>

typedef struct NVRAM_Driver NVRAM_Driver_t;

struct NVRAM_Driver {
    int (*init)(void *driver_ctx);
    int (*read)(void *driver_ctx, uint32_t addr, uint8_t *dest, size_t len);
    int (*write)(void *driver_ctx, uint32_t addr, const uint8_t *src, size_t len);
    int (*erase_chip)(void *driver_ctx);
};

/* Generic application API agnostic to physical bus */
static inline int nvram_read_record(const NVRAM_Driver_t *driver, void *ctx, 
                                    uint32_t addr, uint8_t *data, size_t size) {
    return driver->read(ctx, addr, data, size);
}

9.4 Real-Time Automotive Case: CAN Bus ID Filter & Callback Engine

In automotive ECUs, hundreds of CAN messages arrive per second. The firmware subscribes to specific 11-bit or 29-bit CAN IDs using a binary search callback registry:

Ccan_message_router.c
#include <stdint.h>
#include <stddef.h>

typedef struct {
    uint32_t id;       /* CAN Arb ID (e.g. 0x18FEE000) */
    uint8_t  dlc;      /* Data length code */
    uint8_t  data[8];
} CAN_Frame_t;

typedef void (*CAN_Subscriber_t)(const CAN_Frame_t *frame);

typedef struct {
    uint32_t         can_id;
    CAN_Subscriber_t callback;
} CAN_Route_t;

static void on_engine_speed(const CAN_Frame_t *frame) {
    uint16_t rpm_raw = (frame->data[0] << 8) | frame->data[1];
    /* Process RPM */
}

static void on_brake_pedal(const CAN_Frame_t *frame) {
    /* Process Brake Switch */
}

static const CAN_Route_t ROUTE_TABLE[] = {
    { 0x100UL, on_engine_speed },
    { 0x205UL, on_brake_pedal  }
};

void can_rx_interrupt_handler(const CAN_Frame_t *rx_frame) {
    for (size_t i = 0; i < (sizeof(ROUTE_TABLE)/sizeof(ROUTE_TABLE[0])); ++i) {
        if (ROUTE_TABLE[i].can_id == rx_frame->id) {
            ROUTE_TABLE[i].callback(rx_frame);
            break;
        }
    }
}
